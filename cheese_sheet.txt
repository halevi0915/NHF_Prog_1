def create_screen():
    global screen,height,width
    screen = curses.initscr()
    height,width = screen.getmaxyx()
    screen.clear()
    curses.curs_set(2)
    screen.nodelay(False)
    curses.noecho()

def line(row):
    t = ""
    t += "--|"
    t += "-"*(width-6)
    t += "|--"
    screen.addstr(row,0,t)

def middle(row,message):
    screen.addstr(row, int(width / 2) - int(len(message) / 2), message)

def side(row,message1,message2):
    screen.addstr(row, 3, message1)
    screen.addstr(row,width-(3+len(message2)),message2)
        
def draw_base():
    line(1)
    line(height-2)
    for i in range(0,height):
        screen.addstr(i,2,"|")
    for i in range(0,height):
        screen.addstr(i,width-3,"|")




def draw_decorator(func):
    
    def wrapper(*x):
        screen.clear()
        draw_base()
        func(*x)
        middle(height-1,"")
        screen.refresh()
    return wrapper


@draw_decorator
def main_menu_draw(x):
    middle(3,"IMMORTAL COMBAT")
    middle(4,"by:Profit")
    middle(5,"v8")
    line(height-5)
    side(height-4," > START","") if x%2 == 0 else side(height-4,"   START","")
    side(height-3," > QUIT","") if x%2 == 1 else side(height-3,"   QUIT","")
    #middle(height-1,f"{height}p - {width}p")

def main_menu():
    x = 0
    main_menu_draw(x)
    while True:
        key = screen.getch()
        if key != -1:
            if chr(key) == 'w': x -= 1
            if chr(key) == 's': x += 1
            if chr(key) == '\n': return x%2
        main_menu_draw(x)


curses.endwin()
